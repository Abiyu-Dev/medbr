# Milestone 5 — Transfers, Delivery Coordination, and Supply Gap Reports

Goal: The operational workflow. A facility can request stock from another facility. Transfers move through a strict state machine — every state transition is atomic, locks the source batch, and produces ledger movements only at dispatch/receipt. Phone-based delivery coordination is documented as a call log. Shortages that can't be resolved by transfers become `SupplyGapReport` records for central escalation.

```python
class TransferRequest(models.Model):
    """
    A batch-level stock transfer between two facilities.
    One product / one batch per request (MVP).
    Dispatch and receipt are the ONLY operations that write stock ledger movements.
    """

    class Status(models.TextChoices):
        DRAFT = "DRAFT", "Draft"
        REQUESTED = "REQUESTED", "Requested"
        UNDER_REVIEW = "UNDER_REVIEW", "Under Review"
        APPROVED = "APPROVED", "Approved"
        REJECTED = "REJECTED", "Rejected"
        DISPATCHED = "DISPATCHED", "Dispatched"
        IN_TRANSIT = "IN_TRANSIT", "In Transit"
        RECEIVED = "RECEIVED", "Received"
        CANCELLED = "CANCELLED", "Cancelled"
        FAILED = "FAILED", "Failed"

    class Priority(models.TextChoices):
        CRITICAL = "CRITICAL", "Critical"
        HIGH = "HIGH", "High"
        MEDIUM = "MEDIUM", "Medium"
        LOW = "LOW", "Low"

    source_facility = models.ForeignKey(
        "facilities.Facility",
        on_delete=models.PROTECT,
        related_name="transfers_out",
    )
    destination_facility = models.ForeignKey(
        "facilities.Facility",
        on_delete=models.PROTECT,
        related_name="transfers_in",
    )
    product = models.ForeignKey(
        Product, on_delete=models.PROTECT, related_name="transfers"
    )
    source_batch = models.ForeignKey(
        StockBatch, on_delete=models.PROTECT, related_name="transfers"
    )

    quantity_requested = models.PositiveIntegerField()
    quantity_approved = models.PositiveIntegerField(default=0)
    quantity_received = models.PositiveIntegerField(null=True, blank=True)

    status = models.CharField(
        max_length=16, choices=Status.choices, default=Status.REQUESTED
    )
    priority = models.CharField(
        max_length=16, choices=Priority.choices, default=Priority.MEDIUM
    )

    requested_by = models.ForeignKey(
        "accounts.User",
        on_delete=models.SET_NULL,
        null=True,
        related_name="transfers_requested",
    )
    approved_by = models.ForeignKey(
        "accounts.User",
        on_delete=models.SET_NULL,
        null=True,
        blank=True,
        related_name="transfers_approved",
    )
    approved_at = models.DateTimeField(null=True, blank=True)
    rejection_reason = models.TextField(blank=True)

    received_by = models.ForeignKey(
        "accounts.User",
        on_delete=models.SET_NULL,
        null=True,
        blank=True,
        related_name="transfers_received",
    )
    received_at = models.DateTimeField(null=True, blank=True)
    condition_on_receipt = models.CharField(max_length=32, blank=True)

    expected_dispatch_at = models.DateTimeField(null=True, blank=True)
    expected_arrival_at = models.DateTimeField(null=True, blank=True)

    notes = models.TextField(blank=True)

    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)

    class Meta:
        ordering = ["-created_at"]
        indexes = [
            models.Index(fields=["status", "destination_facility"]),
            models.Index(fields=["status", "source_facility"]),
        ]

    def __str__(self):
        return (
            f"Transfer #{self.id}: {self.source_facility_id} -> "
            f"{self.destination_facility_id} ({self.status})"
        )

    TRANSITIONS = {
        Status.DRAFT: {Status.REQUESTED, Status.CANCELLED},
        Status.REQUESTED: {
            Status.UNDER_REVIEW,
            Status.APPROVED,
            Status.REJECTED,
            Status.CANCELLED,
        },
        Status.UNDER_REVIEW: {Status.APPROVED, Status.REJECTED, Status.CANCELLED},
        Status.APPROVED: {Status.DISPATCHED, Status.CANCELLED},
        Status.DISPATCHED: {Status.IN_TRANSIT, Status.RECEIVED, Status.FAILED},
        Status.IN_TRANSIT: {Status.RECEIVED, Status.FAILED},
        Status.RECEIVED: set(),
        Status.REJECTED: set(),
        Status.CANCELLED: set(),
        Status.FAILED: set(),
    }

    def can_transition_to(self, target):
        return target in self.TRANSITIONS.get(self.status, set())
```

```python
from django.db import transaction
from django.utils import timezone

from .models import StockBatch, StockMovement, TransferRequest


class TransferError(Exception):
    """Raised when a transfer rule is violated. Mapped to HTTP 400."""


# ---------------------------------------------------------------------------
# Creation
# ---------------------------------------------------------------------------


@transaction.atomic
def create_transfer_request(
    *,
    source_facility,
    destination_facility,
    product,
    source_batch,
    quantity_requested,
    priority,
    requested_by,
    expected_dispatch_at=None,
    expected_arrival_at=None,
    notes="",
):
    if source_facility.id == destination_facility.id:
        raise TransferError("Source and destination facilities must differ.")
    for fac in (source_facility, destination_facility):
        if fac.status != "APPROVED" or not fac.is_active:
            raise TransferError(
                f"Facility '{fac.name}' is not approved and active."
            )

    batch = StockBatch.objects.select_for_update().get(pk=source_batch.pk)
    if batch.facility_id != source_facility.id:
        raise TransferError("Source batch does not belong to the source facility.")
    if batch.product_id != product.id:
        raise TransferError("Product does not match the source batch's product.")
    if not batch.is_usable():
        raise TransferError(
            f"Source batch is not usable (status={batch.storage_status}, "
            f"expiry={batch.expiry_date})."
        )
    if quantity_requested <= 0:
        raise TransferError("quantity_requested must be positive.")

    free = batch.quantity_available - batch.quantity_reserved
    if quantity_requested > free:
        raise TransferError(
            f"Insufficient unreserved stock: only {free} free units in source batch."
        )

    return TransferRequest.objects.create(
        source_facility=source_facility,
        destination_facility=destination_facility,
        product=product,
        source_batch=batch,
        quantity_requested=quantity_requested,
        priority=priority,
        requested_by=requested_by,
        expected_dispatch_at=expected_dispatch_at,
        expected_arrival_at=expected_arrival_at,
        notes=notes,
        status=TransferRequest.Status.REQUESTED,
    )


# ---------------------------------------------------------------------------
# Review
# ---------------------------------------------------------------------------


@transaction.atomic
def approve_transfer(*, transfer, approved_by, quantity_approved):
    locked = TransferRequest.objects.select_for_update().get(pk=transfer.pk)
    if locked.status not in (
        TransferRequest.Status.REQUESTED,
        TransferRequest.Status.UNDER_REVIEW,
    ):
        raise TransferError(f"Cannot approve a transfer in status {locked.status}.")
    if quantity_approved <= 0:
        raise TransferError("quantity_approved must be positive.")
    if quantity_approved > locked.quantity_requested:
        raise TransferError("quantity_approved cannot exceed quantity_requested.")

    batch = StockBatch.objects.select_for_update().get(pk=locked.source_batch_id)
    if not batch.is_usable():
        raise TransferError("Source batch is no longer usable.")
    free = batch.quantity_available - batch.quantity_reserved
    if quantity_approved > free:
        raise TransferError(
            f"Insufficient free stock in source batch: {free} available, "
            f"{quantity_approved} requested."
        )

    batch.quantity_reserved += quantity_approved
    batch.save(update_fields=["quantity_reserved", "updated_at"])

    locked.quantity_approved = quantity_approved
    locked.status = TransferRequest.Status.APPROVED
    locked.approved_by = approved_by
    locked.approved_at = timezone.now()
    locked.save()
    return locked


@transaction.atomic
def reject_transfer(*, transfer, reviewed_by, rejection_reason):
    if not rejection_reason or not rejection_reason.strip():
        raise TransferError("Rejection reason is required.")
    locked = TransferRequest.objects.select_for_update().get(pk=transfer.pk)
    if locked.status not in (
        TransferRequest.Status.REQUESTED,
        TransferRequest.Status.UNDER_REVIEW,
    ):
        raise TransferError(f"Cannot reject a transfer in status {locked.status}.")

    locked.status = TransferRequest.Status.REJECTED
    locked.rejection_reason = rejection_reason
    locked.approved_by = reviewed_by  # reviewer field doubles for rejections
    locked.approved_at = timezone.now()
    locked.save()
    return locked


# ---------------------------------------------------------------------------
# Dispatch
# ---------------------------------------------------------------------------


@transaction.atomic
def dispatch_transfer(*, transfer, performed_by, notes=""):
    locked = TransferRequest.objects.select_for_update().get(pk=transfer.pk)
    if locked.status != TransferRequest.Status.APPROVED:
        raise TransferError(
            f"Cannot dispatch a transfer in status {locked.status}."
        )

    batch = StockBatch.objects.select_for_update().get(pk=locked.source_batch_id)
    if batch.quantity_available < locked.quantity_approved:
        raise TransferError(
            f"Source batch only has {batch.quantity_available} units; "
            f"{locked.quantity_approved} are committed."
        )

    batch.quantity_available -= locked.quantity_approved
    batch.quantity_reserved = max(
        0, batch.quantity_reserved - locked.quantity_approved
    )
    batch.save(update_fields=["quantity_available", "quantity_reserved", "updated_at"])

    StockMovement.objects.create(
        batch=batch,
        movement_type=StockMovement.MovementType.TRANSFER_OUT,
        direction=StockMovement.Direction.OUT,
        quantity=locked.quantity_approved,
        reference_type="TransferRequest",
        reference_id=str(locked.id),
        reason=notes or "Dispatched transfer",
        performed_by=performed_by,
        idempotency_key=f"transfer-{locked.id}-out",
    )

    locked.status = TransferRequest.Status.DISPATCHED
    locked.save()
    return locked


# ---------------------------------------------------------------------------
# Receipt
# ---------------------------------------------------------------------------


@transaction.atomic
def receive_transfer(*, transfer, performed_by, quantity_received, condition, notes=""):
    locked = TransferRequest.objects.select_for_update().get(pk=transfer.pk)
    if locked.status not in (
        TransferRequest.Status.DISPATCHED,
        TransferRequest.Status.IN_TRANSIT,
    ):
        raise TransferError(
            f"Cannot receive a transfer in status {locked.status}."
        )
    if quantity_received <= 0:
        raise TransferError("quantity_received must be positive.")
    if quantity_received > locked.quantity_approved:
        raise TransferError(
            f"quantity_received ({quantity_received}) exceeds quantity_approved "
            f"({locked.quantity_approved})."
        )

    condition = (condition or "OK").upper()
    allowed_conditions = {"OK", "DAMAGED", "TEMPERATURE_EXCURSION"}
    if condition not in allowed_conditions:
        raise TransferError(f"Invalid condition '{condition}'.")

    # Destination batch: preserve source batch_number and expiry for traceability
    existing = (
        StockBatch.objects.select_for_update()
        .filter(
            facility=locked.destination_facility,
            product=locked.product,
            batch_number=locked.source_batch.batch_number,
        )
        .first()
    )

    if existing is not None:
        if existing.expiry_date != locked.source_batch.expiry_date:
            raise TransferError(
                f"Destination already holds batch '{existing.batch_number}' with "
                f"expiry {existing.expiry_date}; source batch expiry is "
                f"{locked.source_batch.expiry_date}. Cannot merge mismatched batches."
            )
        if existing.storage_status != StockBatch.StorageStatus.OK and condition == "OK":
            raise TransferError(
                f"Destination batch is currently {existing.storage_status}; "
                "cannot receive into a non-OK batch with condition OK."
            )
        existing.quantity_available += quantity_received
        existing.quantity_received += quantity_received
        if condition != "OK":
            existing.storage_status = condition
        existing.save(
            update_fields=[
                "quantity_available",
                "quantity_received",
                "storage_status",
                "updated_at",
            ]
        )
        dest_batch = existing
    else:
        dest_batch = StockBatch.objects.create(
            facility=locked.destination_facility,
            product=locked.product,
            batch_number=locked.source_batch.batch_number,
            expiry_date=locked.source_batch.expiry_date,
            quantity_received=quantity_received,
            quantity_available=quantity_received,
            storage_status=condition if condition != "OK" else "OK",
            source_reference=f"Transfer #{locked.id}",
            created_by=performed_by,
        )

    StockMovement.objects.create(
        batch=dest_batch,
        movement_type=StockMovement.MovementType.TRANSFER_IN,
        direction=StockMovement.Direction.IN,
        quantity=quantity_received,
        reference_type="TransferRequest",
        reference_id=str(locked.id),
        reason=notes or "Received transfer",
        performed_by=performed_by,
        idempotency_key=f"transfer-{locked.id}-in",
    )

    locked.quantity_received = quantity_received
    locked.received_by = performed_by
    locked.received_at = timezone.now()
    locked.condition_on_receipt = condition
    locked.status = TransferRequest.Status.RECEIVED
    locked.save()

    return locked, dest_batch


# ---------------------------------------------------------------------------
# Cancel
# ---------------------------------------------------------------------------


@transaction.atomic
def cancel_transfer(*, transfer, performed_by, reason=""):
    locked = TransferRequest.objects.select_for_update().get(pk=transfer.pk)
    if locked.status not in (
        TransferRequest.Status.REQUESTED,
        TransferRequest.Status.UNDER_REVIEW,
        TransferRequest.Status.APPROVED,
    ):
        raise TransferError(f"Cannot cancel a transfer in status {locked.status}.")

    if locked.status == TransferRequest.Status.APPROVED:
        batch = StockBatch.objects.select_for_update().get(pk=locked.source_batch_id)
        batch.quantity_reserved = max(
            0, batch.quantity_reserved - locked.quantity_approved
        )
        batch.save(update_fields=["quantity_reserved", "updated_at"])

    locked.status = TransferRequest.Status.CANCELLED
    if reason:
        locked.notes = (locked.notes + "\n" if locked.notes else "") + f"Cancelled: {reason}"
    locked.save()
    return locked
```

```python
def get_usable_stock(facility, product, as_of=None):
    """
    Stock that is physically OK, not expired, and not committed to another
    approved transfer (i.e. available for allocation).
    """
    as_of = as_of or timezone.localdate()
    batches = StockBatch.objects.filter(
        facility=facility,
        product=product,
        storage_status=StockBatch.StorageStatus.OK,
        expiry_date__gte=as_of,
        quantity_available__gt=0,
    )
    return sum(
        max(0, b.quantity_available - b.quantity_reserved) for b in batches
    )


def get_incoming_stock(facility, product, horizon_days=30, as_of=None):
    """
    Confirmed (approved/dispatched/in-transit) transfers headed to this facility.
    Unapproved or rejected transfers are NOT counted.
    """
    from .models import TransferRequest

    as_of = as_of or timezone.localdate()
    horizon_end = as_of + timedelta(days=horizon_days)

    qs = TransferRequest.objects.filter(
        destination_facility=facility,
        product=product,
        status__in=[
            TransferRequest.Status.APPROVED,
            TransferRequest.Status.DISPATCHED,
            TransferRequest.Status.IN_TRANSIT,
        ],
    ).filter(
        models.Q(expected_arrival_at__isnull=True)
        | models.Q(expected_arrival_at__date__lte=horizon_end)
    )
    return int(qs.aggregate(total=Sum("quantity_approved"))["total"] or 0)
```

```python
class SupplyGapReport(models.Model):
    """
    One *active* gap per (facility, product). Terminal reports (RESOLVED/CLOSED)
    are immutable; a new gap that re-emerges creates a new report.
    """

    class Priority(models.TextChoices):
        CRITICAL = "CRITICAL", "Critical"
        HIGH = "HIGH", "High"
        MEDIUM = "MEDIUM", "Medium"
        LOW = "LOW", "Low"

    class Status(models.TextChoices):
        OPEN = "OPEN", "Open"
        UNDER_REVIEW = "UNDER_REVIEW", "Under Review"
        ASSIGNED = "ASSIGNED", "Assigned"
        ACTION_PLANNED = "ACTION_PLANNED", "Action Planned"
        ESCALATED = "ESCALATED", "Escalated"
        RESOLVED = "RESOLVED", "Resolved"
        CLOSED = "CLOSED", "Closed"

    NON_TERMINAL_STATUSES = [
        Status.OPEN,
        Status.UNDER_REVIEW,
        Status.ASSIGNED,
        Status.ACTION_PLANNED,
        Status.ESCALATED,
    ]
    TERMINAL_STATUSES = [Status.RESOLVED, Status.CLOSED]

    facility = models.ForeignKey(
        "facilities.Facility",
        on_delete=models.PROTECT,
        related_name="supply_gaps",
    )
    product = models.ForeignKey(
        "inventory.Product",
        on_delete=models.PROTECT,
        related_name="supply_gaps",
    )
    reporting_horizon_start = models.DateField()
    reporting_horizon_end = models.DateField()
    projected_demand = models.IntegerField()
    usable_stock = models.IntegerField()
    reliable_incoming_quantity = models.IntegerField(default=0)
    uncovered_quantity = models.IntegerField()
    priority = models.CharField(max_length=16, choices=Priority.choices)
    reason_summary = models.TextField(blank=True)
    status = models.CharField(
        max_length=16, choices=Status.choices, default=Status.OPEN
    )
    assigned_to = models.ForeignKey(
        "accounts.User",
        on_delete=models.SET_NULL,
        null=True,
        blank=True,
        related_name="assigned_supply_gaps",
    )
    resolution_notes = models.TextField(blank=True)
    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)
    resolved_at = models.DateTimeField(null=True, blank=True)

    class Meta:
        ordering = ["priority", "-uncovered_quantity", "-created_at"]
        indexes = [
            models.Index(fields=["status", "priority"]),
            models.Index(fields=["facility", "product"]),
        ]
        constraints = [
            models.UniqueConstraint(
                fields=["facility", "product"],
                condition=models.Q(
                    status__in=[
                        "OPEN",
                        "UNDER_REVIEW",
                        "ASSIGNED",
                        "ACTION_PLANNED",
                        "ESCALATED",
                    ]
                ),
                name="unique_open_gap_per_facility_product",
            )
        ]

    def __str__(self):
        return (
            f"Gap {self.facility_id}/{self.product_id} "
            f"({self.priority}, qty={self.uncovered_quantity})"
        )

    def save(self, *args, **kwargs):
        if self.status in (self.Status.RESOLVED, self.Status.CLOSED):
            if self.resolved_at is None:
                self.resolved_at = timezone.now()
        else:
            self.resolved_at = None
        super().save(*args, **kwargs)
```

```python
from datetime import date, timedelta

from django.db import transaction
from django.utils import timezone

from inventory.calculations import get_product_summary
from inventory.models import Product, StockBatch

from .models import SupplyGapReport


def _derive_priority(days_of_stock, projected_gap, projected_demand):
    if projected_gap is None or projected_gap <= 0:
        return None
    if days_of_stock is None or days_of_stock <= 0:
        return SupplyGapReport.Priority.CRITICAL
    if days_of_stock < 7:
        return SupplyGapReport.Priority.CRITICAL
    if days_of_stock < 14:
        return SupplyGapReport.Priority.HIGH
    if days_of_stock < 30:
        return SupplyGapReport.Priority.MEDIUM
    return SupplyGapReport.Priority.LOW


def _build_reason_summary(summary):
    parts = [
        f"Projected demand {summary['projected_demand']} over horizon;",
        f"usable stock {summary['usable_stock']};",
        f"reliable incoming {summary['incoming_stock']};",
        f"uncovered {summary['projected_gap']}.",
    ]
    if summary["days_of_stock"] is not None:
        parts.append(f"Days of stock: {summary['days_of_stock']}.")
    return " ".join(parts)


@transaction.atomic
def upsert_supply_gap(
    *,
    facility,
    product,
    horizon_days=30,
    usage_window_days=30,
    summary=None,
):
    """
    Create or update the single active gap report for (facility, product).
    Returns (report_or_None, created_bool).
    """
    if summary is None:
        summary = get_product_summary(
            facility,
            product,
            horizon_days=horizon_days,
            usage_window_days=usage_window_days,
        )

    priority = _derive_priority(
        summary["days_of_stock"],
        summary["projected_gap"],
        summary["projected_demand"],
    )
    if priority is None:
        return None, False

    today = timezone.localdate()
    horizon_end = today + timedelta(days=horizon_days)

    existing = (
        SupplyGapReport.objects.select_for_update()
        .filter(
            facility=facility,
            product=product,
            status__in=SupplyGapReport.NON_TERMINAL_STATUSES,
        )
        .first()
    )

    if existing:
        existing.reporting_horizon_start = today
        existing.reporting_horizon_end = horizon_end
        existing.projected_demand = summary["projected_demand"]
        existing.usable_stock = summary["usable_stock"]
        existing.reliable_incoming_quantity = summary["incoming_stock"]
        existing.uncovered_quantity = summary["projected_gap"]
        existing.priority = priority
        existing.reason_summary = _build_reason_summary(summary)
        existing.save()
        return existing, False

    report = SupplyGapReport.objects.create(
        facility=facility,
        product=product,
        reporting_horizon_start=today,
        reporting_horizon_end=horizon_end,
        projected_demand=summary["projected_demand"],
        usable_stock=summary["usable_stock"],
        reliable_incoming_quantity=summary["incoming_stock"],
        uncovered_quantity=summary["projected_gap"],
        priority=priority,
        reason_summary=_build_reason_summary(summary),
        status=SupplyGapReport.Status.OPEN,
    )
    return report, True


def refresh_supply_gaps(*, facility=None, horizon_days=30, usage_window_days=30):
    """
    Iterate every (facility, product) that has batches and upsert its open gap.
    Returns (created, updated).
    """
    from facilities.models import Facility

    facilities = Facility.objects.filter(status="APPROVED", is_active=True)
    if facility is not None:
        facilities = facilities.filter(pk=facility.pk)

    created_count = 0
    updated_count = 0
    for fac in facilities:
        product_ids = set(
            StockBatch.objects.filter(facility=fac).values_list(
                "product_id", flat=True
            )
        )
        products = Product.objects.filter(id__in=product_ids, active=True)
        for product in products:
            report, was_created = upsert_supply_gap(
                facility=fac,
                product=product,
                horizon_days=horizon_days,
                usage_window_days=usage_window_days,
            )
            if report is None:
                continue
            if was_created:
                created_count += 1
            else:
                updated_count += 1
    return created_count, updated_count
```

```python
                "updated": updated,

    def assign(self, request, pk=None):

        ser = SupplyGapAssignSerializer(data=request.data)
        ser.is_valid(raise_exception=True)

        try:
            user = User.objects.get(pk=ser.validated_data["assigned_to"])
```

```python
                    status=status.HTTP_403_FORBIDDEN,
                )

        ser = TransferApproveSerializer(data=request.data)
        ser.is_valid(raise_exception=True)
        qty = ser.validated_data.get("quantity_approved", transfer.quantity_requested)

        try:
            transfer = approve_transfer(
                transfer=transfer, approved_by=user, quantity_approved=qty
            )
        except TransferError as exc:
            if "Cannot approve" in str(exc):
                return _transfer_state_conflict(exc)
            return _transfer_error_response(exc)

        log_audit(
            user,
            "TRANSFER_APPROVED",
            "TransferRequest",
            transfer.id,
            metadata={"quantity_approved": qty},
            ip_address=_ip(request),
        )
        return Response(TransferRequestSerializer(transfer).data)

    # -------------------- Reject --------------------

    @action(detail=True, methods=["post"])
    def reject(self, request, pk=None):
        transfer = self.get_object()
        user = request.user
        is_central = user.role in (
            User.Role.SYSTEM_ADMIN,
            User.Role.CENTRAL_COORDINATOR,
        )
        if not is_central:
            if not (
                user.role == User.Role.FACILITY_MANAGER
                and user.facility_id
                in (transfer.source_facility_id, transfer.destination_facility_id)
            ):
                return Response(
                    {"detail": "Not authorized.", "code": "forbidden"},
                    status=status.HTTP_403_FORBIDDEN,
                )

        ser = TransferRejectSerializer(data=request.data)
        ser.is_valid(raise_exception=True)

        try:
            transfer = reject_transfer(
                transfer=transfer,
                reviewed_by=user,
                rejection_reason=ser.validated_data["rejection_reason"],
            )
        except TransferError as exc:
            if "Cannot reject" in str(exc):
                return _transfer_state_conflict(exc)
            return _transfer_error_response(exc)

        log_audit(
            user,
            "TRANSFER_REJECTED",
            "TransferRequest",
            transfer.id,
            metadata={"reason": transfer.rejection_reason},
            ip_address=_ip(request),
        )
        return Response(TransferRequestSerializer(transfer).data)

    # -------------------- Dispatch --------------------

    @action(detail=True, methods=["post"])
    def dispatch(self, request, pk=None):
        transfer = self.get_object()
        user = request.user
        is_central = user.role in (
            User.Role.SYSTEM_ADMIN,
            User.Role.CENTRAL_COORDINATOR,
        )
        if not is_central:
            if not (

        except TransferError as exc:

            return _transfer_error_response(exc)

        log_audit(
            user,
            "TRANSFER_DISPATCHED",
            "TransferRequest",
            transfer.id,
            metadata={"quantity": transfer.quantity_approved},
            ip_address=_ip(request),
        )
        return Response(TransferRequestSerializer(transfer).data)

    # -------------------- Receive --------------------

    @action(detail=True, methods=["post"], url_path="confirm-receipt")
    def confirm_receipt(self, request, pk=None):
        transfer = self.get_object()

            User.Role.SYSTEM_ADMIN,

                    },
                    status=status.HTTP_403_FORBIDDEN,
                )

        ser.is_valid(raise_exception=True)

        try:
            transfer, dest_batch = receive_transfer(
                transfer=transfer,
                performed_by=user,
                quantity_received=ser.validated_data["quantity_received"],
                condition=ser.validated_data["condition"],
                notes=ser.validated_data.get("notes", ""),
            )
        except TransferError as exc:
            if "Cannot receive" in str(exc):
                return _transfer_state_conflict(exc)
            return _transfer_error_response(exc)

        log_audit(
            user,
            "TRANSFER_RECEIVED",
            "TransferRequest",
            transfer.id,
            metadata={
                "quantity_received": transfer.quantity_received,
                "condition": transfer.condition_on_receipt,
                "destination_batch_id": dest_batch.id,
            },
            ip_address=_ip(request),
        )

        if transfer.quantity_received < transfer.quantity_approved:
            log_audit(
                user,
                "TRANSFER_DISCREPANCY",
                "TransferRequest",
                transfer.id,
                metadata={
                    "approved": transfer.quantity_approved,
                    "received": transfer.quantity_received,
                    "delta": transfer.quantity_approved
                    - transfer.quantity_received,
                },
                ip_address=_ip(request),
            )

        return Response(TransferRequestSerializer(transfer).data)
```

## Step 19 — What was intentionally NOT built in M5

- No `TransferItem` — one product/batch per transfer in MVP (docs explicitly allow this).
- No **auto-triggered supply gap refresh** — `POST /central/supply-gaps/refresh/` is explicit. M8's AI summarizer will call `refresh_supply_gaps()` internally.
- No **OSRM routing integration** — `expected_arrival_at` is set by the coordinator manually or via a future routing provider. Docs §4.2 say routing is an estimate; not required for the state machine.
- No **email/SMS notification** — deferred per docs.
- No **reservation release on `EXPIRED` batch** — if a batch expires while reserved, the next `dispatch` will fail the `is_usable()` check. Acceptable for MVP.
- No **partial-quantity shipping** — dispatch always ships `quantity_approved`; receipt may be less but never more.
- No **driver entity** — coordination is a call log only.
- No **cross-transfer batching** — one route per transfer.
- **Deviation from literal API list**: I added `POST /central/supply-gaps/refresh/` (not in the docs list) as the mechanism to (re)generate reports deterministically without AI. Without it there'd be no way to populate the table for the MVP demo — the alternative would be a background job we don't have. Documented here.
